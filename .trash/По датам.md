```dataviewjs
// Настройки
const folder = "_posts";          // папка с постами
const dateField = "date";         // поле в frontmatter с датой

// Текущий месяц и год
const now = new Date();
const month = now.getMonth();     // 0-11
const year = now.getFullYear();

// Заголовок
const monthNames = ["Январь","Февраль","Март","Апрель","Май","Июнь","Июль","Август","Сентябрь","Октябрь","Ноябрь","Декабрь"];
dv.header(2, `${monthNames[month]} ${year}`);

// Получаем все файлы из папки, у которых есть поле date
const files = dv.pages(`"${folder}"`)
  .where(p => p[dateField]);

// Строим карту: дата (YYYY-MM-DD) -> ссылка на файл
const dateMap = new Map();
files.forEach(f => {
  let d = f[dateField];
  if (d) {
    if (typeof d === 'object' && d.isMoment) {
      d = d.format('YYYY-MM-DD');
    } else if (typeof d === 'string') {
      d = d.substring(0, 10);
    } else {
      return;
    }
    dateMap.set(d, f.file.link);
  }
});

// Построение календарной сетки
const daysOfWeek = ["Вс", "Пн", "Вт", "Ср", "Чт", "Пт", "Сб"];
const firstDay = new Date(year, month, 1);
const lastDay = new Date(year, month + 1, 0);
const daysInMonth = lastDay.getDate();
const startDayOfWeek = firstDay.getDay(); // 0 - воскресенье

// Формируем строки таблицы (каждая строка - массив из 7 элементов)
const rows = [];
let currentRow = [];

// Пустые ячейки до первого дня
for (let i = 0; i < startDayOfWeek; i++) {
  currentRow.push(" ");
}

// Заполняем дни месяца
for (let d = 1; d <= daysInMonth; d++) {
  const dateStr = `${year}-${String(month+1).padStart(2,'0')}-${String(d).padStart(2,'0')}`;
  const link = dateMap.get(dateStr);
  // Если есть заметка – вставляем ссылку, иначе – просто число
  currentRow.push(link ? link : d.toString());
  
  if (currentRow.length === 7) {
    rows.push(currentRow);
    currentRow = [];
  }
}

// Если остались ячейки в последней строке, дополняем пробелами
if (currentRow.length > 0) {
  while (currentRow.length < 7) currentRow.push(" ");
  rows.push(currentRow);
}

// Выводим таблицу с днями недели в заголовке
dv.table(daysOfWeek, rows);
```
